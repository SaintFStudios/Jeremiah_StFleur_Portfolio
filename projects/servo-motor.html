Custom 360° Positioning Servo Motor

Closed-Loop Actuator from Component Selection to Control
Nov 2024 – Dec 2024 Robotics & Mechanical Design
Actuator Design PID Control Gear Design 3D Printing System Integration

Project Overview

Design and fabrication of a high-precision, continuous-rotation robotic actuator from component selection to closed-loop control. The actuator features a fully 3D-printed housing with integrated mounting, a magnetic encoder for full 360° position sensing, and a potentiometer-driven setpoint with an exponential smoothing filter for deceleration behavior. The gear train was later upgraded from spur to helical gears, improving efficiency, reducing backlash, and quieting operation significantly.

Position_Control.ino

Proportional closed-loop controller — potentiometer setpoint mapped to 360°, exponential smoothing for deceleration, shortest-path error resolution for CW/CCW selection

Position_Control.ino C++ / Arduino
// pins
int ENA = 2;            // enable pin
int pwm[2] = { 5, 6 };  // motor driver pwm pins
int encoder = A0;       // encoder pin
int pot = A1;
float Pot_Value = 0, offset = 0;
float position, Prev_position, Position_Smoothed;
 
void setup() {
  pinMode(ENA, OUTPUT);
  pinMode(pot, INPUT);
  Serial.begin(115200);
}
 
void loop() {
  Pot_Value = analogRead(pot);
  position = map(Pot_Value, 0, 1023, 0, 360);
  // exponential smoothing — servo decelerates
  Position_Smoothed = (position * 0.05) + (Prev_position * 0.95);
  Prev_position = Position_Smoothed;
  moveTo(Position_Smoothed);
  Serial.println(Position_Smoothed);
  delay(15);
}
 
float moveTo(float setpoint) {
  float error, out;
  // shortest-path error resolution
  float CW_error  = setpoint - getPos();
  if (CW_error < 0) CW_error += 360;
  float CCW_error = getPos() - setpoint;
  if (CCW_error < 0) CCW_error += 360;
  if (CW_error <= CCW_error) error = CW_error;
  else error = -1 * CCW_error;
 
  out = 1.5 * error;
  out = constrain(out, -255, 255);
  if (abs(out) < 25) out = 0;
 
  digitalWrite(ENA, HIGH);
  if (out > 0) {
    analogWrite(pwm[0], 0); analogWrite(pwm[1], out);
  } else if (out < 0) {
    analogWrite(pwm[0], abs(out)); analogWrite(pwm[1], 0);
  } else {
    analogWrite(pwm[0], 0); analogWrite(pwm[1], 0);
  }
}
 
float getPos() {
  float pos = map(analogRead(encoder), 0, 1023, 0, 359) + offset;
  if (pos < 0) pos = 359 + pos;
  else if (pos > 359) pos = pos - 359;
  return pos;
}
